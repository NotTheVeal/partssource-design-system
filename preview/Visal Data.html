Nivo Theme Config

Drop-in psNivoTheme — pass as theme prop to every Nivo chart
export const psNivoTheme = {
  background: 'transparent',
  textColor: '#4A4A4A',            // --ps-fg-primary
  fontSize: 12,
  fontFamily: "'Source Sans Pro', sans-serif",
  axis: {
    ticks:  { text: { fill: '#777777', fontSize: 11 } },  // --ps-fg-secondary
    legend: { text: { fill: '#4A4A4A', fontSize: 12, fontWeight: 600 } }
  },
  grid: { line: { stroke: '#DCDCDC', strokeWidth: 1 } },  // --ps-border-default
  legends: { text: { fill: '#4A4A4A', fontSize: 11 } },
  tooltip: {
    container: {
      background: '#FFFFFF', color: '#4A4A4A', fontSize: 12,
      borderRadius: 4, border: '1px solid #DCDCDC',
      boxShadow: '0 2px 10px rgba(0,47,72,0.10)',         // --ps-shadow-md
      fontFamily: "'Source Sans Pro', sans-serif"
    }
  }
};
